  return new Intl.NumberFormat('vi-VN').format(price) + 'đ';
};

    setToast({ show: true, message });

  };

  const addToCart = (item) => {
    setCart((prev) => {
      const existing = prev.find((i) => i.id === item.id);
      if (existing) {
        return prev.map((i) => (i.id === item.id ? { ...i, quantity: i.quantity + 1 } : i));
      }
      return [...prev, { ...item, quantity: 1 }];
    });
    showToast(`Đã thêm ${item.name} vào giỏ! 😋`);
  };

  const updateQuantity = (id, delta) => {
    setCart((prev) =>
      prev.map((item) => {
        if (item.id === id) {
          const newQ = item.quantity + delta;
          return { ...item, quantity: newQ > 0 ? newQ : 1 };
        }
        return item;
      })
    );

                <span className="absolute -top-1 -right-1 bg-emerald-400 text-white text-[11px] font-black h-5 w-5 rounded-full flex items-center justify-center border-2 border-white shadow-sm">

            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">

            <button 
              type="submit"
              className="w-full bg-emerald-400 hover:bg-emerald-500 text-white py-4 rounded-2xl font-bold text-lg shadow-lg shadow-emerald-400/30 transition-all hover:-translate-y-0.5 flex justify-center items-center gap-2"
            >
              Gửi Tâm Tình <Send size={20} />
            </button>
          </form>
        </div>
      </section>

      {/* --- FOOTER --- */}
      <footer className="bg-pink-100 pt-16 pb-8 border-t border-pink-200 text-pink-900">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
          <div className="grid grid-cols-1 md:grid-cols-3 gap-10 md:gap-8 mb-12">
            
            {/* Column 1: About */}
            <div className="space-y-4">
              <h3 className="text-2xl font-black text-pink-600 mb-4 flex items-center gap-2">
                3CE FOODS <Heart size={20} className="fill-pink-500 text-pink-500"/>
              </h3>
              <p className="font-bold text-sm italic">"Ăn vặt không no - Không lo bị péoooo"</p>
              <div className="space-y-3 mt-6 text-sm font-medium">
                <p className="flex items-start gap-3">
                  <MapPin size={18} className="text-pink-500 mt-0.5 flex-shrink-0" />
                  <span>Cung đường ẩm thực số 1, Quận Ăn Uống, TP. Hạnh Phúc</span>
                </p>
                <p className="flex items-center gap-3">
                  <Phone size={18} className="text-pink-500 flex-shrink-0" />
                  <span className="font-bold">Hotline: 1900 xxxx</span>
                </p>
              </div>
            </div>

            {/* Column 2: Policies */}
            <div>
              <h3 className="text-lg font-black text-pink-600 mb-6">Chính Sách</h3>
              <ul className="space-y-3 text-sm font-medium">
                <li><a href="#" className="hover:text-pink-600 hover:underline transition-all">Chính sách chăm sóc khách hàng</a></li>
                <li><a href="#" className="hover:text-pink-600 hover:underline transition-all">Chính sách hoạt động</a></li>
                <li><a href="#" className="hover:text-pink-600 hover:underline transition-all">Chính sách và quy định</a></li>
                <li><a href="#" className="hover:text-pink-600 hover:underline transition-all">Chính sách bảo mật thông tin</a></li>
              </ul>
            </div>

            {/* Column 3: Social */}
            <div>
              <h3 className="text-lg font-black text-pink-600 mb-6">Kết Nối Với Chúng Mình</h3>
              <p className="text-sm font-medium mb-4">Theo dõi để không bỏ lỡ những ưu đãi ngọt ngào nhất!</p>
              <a href="#" className="inline-flex items-center justify-center gap-2 bg-white text-blue-600 px-6 py-3 rounded-2xl font-bold shadow-sm hover:shadow-md transition-all hover:-translate-y-1">
                <Facebook size={20} /> Đường liên kết qua Facebook
              </a>
            </div>
            
          </div>
          
          <div className="text-center text-sm font-medium text-pink-700/70 pt-8 border-t border-pink-200/50">
            &copy; {new Date().getFullYear()} 3CE FOODS. All rights reserved. Designed with ❤️
          </div>
        </div>
      </footer>

      {/* --- CART SIDEBAR OVERLAY --- */}
      <div 
        className={`fixed inset-0 bg-gray-900/30 backdrop-blur-sm z-50 transition-opacity duration-300 ${
          isCartOpen ? 'opacity-100 visible' : 'opacity-0 invisible'
        }`}
        onClick={() => setIsCartOpen(false)}
      ></div>

      {/* --- CART SIDEBAR --- */}
      <div 
        className={`fixed top-0 right-0 h-full w-full sm:w-[450px] bg-white shadow-2xl z-50 flex flex-col transform transition-transform duration-400 ease-in-out ${
          isCartOpen ? 'translate-x-0' : 'translate-x-full'
        }`}
      >
        <div className="flex items-center justify-between p-6 border-b border-pink-100 bg-pink-50/50">
          <h2 className="text-2xl font-black text-pink-600 flex items-center gap-2">
            <ShoppingCart size={24} /> Giỏ Hàng
            <span className="bg-white text-emerald-500 text-sm py-1 px-3 rounded-full font-bold shadow-sm ml-2">
              {totalItems} món
            </span>
          </h2>
          <button 
            onClick={() => setIsCartOpen(false)}
            className="p-2 text-gray-400 hover:bg-pink-100 hover:text-pink-500 rounded-full transition-colors"
          >
            <X size={24} />
          </button>
        </div>

        <div className="flex-1 overflow-y-auto p-4 sm:p-6 bg-gray-50/30">
          {cart.length === 0 ? (
            <div className="h-full flex flex-col items-center justify-center text-gray-400 space-y-4">
              <div className="bg-pink-50 p-6 rounded-full">
                <ShoppingCart size={64} className="text-pink-200" />
              </div>
              <p className="text-lg font-bold text-gray-500">Giỏ hàng trống trơn! 🥺</p>
              <button 
                onClick={() => setIsCartOpen(false)}
                className="text-emerald-500 font-bold hover:underline bg-emerald-50 px-6 py-2 rounded-full"
              >
                Cùng chọn món ngon nào!
              </button>
            </div>
          ) : (
            <div className="space-y-4">
              {cart.map((item) => (
                <div key={item.id} className="flex gap-4 items-center bg-white p-4 rounded-2xl border border-pink-100 shadow-sm">
                  <img src={item.img} alt={item.name} className="w-20 h-20 object-cover rounded-xl shadow-sm" />
                  <div className="flex-1">
                    <h4 className="font-bold text-gray-800 text-sm mb-1 leading-snug">{item.name}</h4>
                    <p className="text-emerald-500 font-black text-sm mb-3">{formatPrice(item.price)}</p>
                    <div className="flex items-center justify-between">
                      <div className="flex items-center bg-pink-50 border border-pink-100 rounded-xl overflow-hidden">
                        <button 
                          onClick={() => updateQuantity(item.id, -1)}
                          className="w-8 h-8 flex items-center justify-center text-pink-600 hover:bg-pink-200 transition-colors"
                        >
                          <Minus size={14} strokeWidth={3} />
                        </button>
                        <span className="w-8 text-center font-bold text-sm text-gray-800">{item.quantity}</span>
                        <button 
                          onClick={() => updateQuantity(item.id, 1)}
                          className="w-8 h-8 flex items-center justify-center text-pink-600 hover:bg-pink-200 transition-colors"
                        >
                          <Plus size={14} strokeWidth={3} />
                        </button>
                      </div>
                      <button 
                        onClick={() => removeFromCart(item.id)}
                        className="p-2 text-gray-300 hover:text-red-500 hover:bg-red-50 rounded-xl transition-colors"
                      >
                        <Trash2 size={18} />
                      </button>
                    </div>
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>

        {cart.length > 0 && (
          <div className="p-6 bg-white border-t border-pink-100 shadow-[0_-10px_40px_rgba(0,0,0,0.05)]">
            <div className="flex justify-between items-center mb-6">
              <span className="text-gray-500 font-bold">Tổng thanh toán:</span>
              <span className="text-3xl font-black text-pink-500">{formatPrice(totalPrice)}</span>
            </div>
            <button 
              onClick={() => {
                showToast("Chuyển đến trang thanh toán... 🚀");
                setIsCartOpen(false);
              }}
              className="w-full bg-pink-500 hover:bg-pink-600 text-white py-4.5 rounded-2xl font-black text-lg shadow-lg shadow-pink-500/30 flex items-center justify-center gap-2 transition-all hover:-translate-y-1 active:translate-y-0"
            >
              Thanh Toán Ngay <ChevronRight size={22} />
            </button>
          </div>
        )}
      </div>

      {/* --- FLOATING CHAT WIDGET --- */}
      <div className="fixed bottom-6 right-6 z-40 flex flex-col items-end">
        {/* Chat Window */}
        <div 
          className={`mb-4 w-80 bg-white rounded-3xl shadow-2xl border border-pink-100 overflow-hidden transition-all duration-300 transform origin-bottom-right ${
            isChatOpen ? 'scale-100 opacity-100' : 'scale-0 opacity-0'
          }`}
        >
          <div className="bg-pink-500 p-4 flex justify-between items-center text-white">
            <div className="flex items-center gap-2 font-bold">
              <MessageCircle size={18} /> 3CE FOODS Hỗ trợ
            </div>
            <button onClick={() => setIsChatOpen(false)} className="hover:bg-pink-600 p-1 rounded-full transition-colors">
              <X size={18} />
            </button>
          </div>
          <div className="h-64 bg-pink-50/30 p-4 overflow-y-auto flex flex-col gap-3">
            {chatMessages.map((msg, idx) => (
              <div key={idx} className={`flex ${msg.sender === 'user' ? 'justify-end' : 'justify-start'}`}>
                <div className={`max-w-[80%] p-3 rounded-2xl text-sm font-medium ${
                  msg.sender === 'user' 
                    ? 'bg-emerald-400 text-white rounded-br-sm' 
                    : 'bg-white border border-pink-100 text-gray-700 rounded-bl-sm shadow-sm'
                }`}>
                  {msg.text}
                </div>
              </div>
            ))}
          </div>
          <form onSubmit={handleSendChat} className="p-3 bg-white border-t border-pink-100 flex gap-2">
            <input 
              type="text" 
              value={currentMessage}
              onChange={(e) => setCurrentMessage(e.target.value)}
              placeholder="Nhập tin nhắn..."
              className="flex-1 bg-gray-50 border border-gray-200 rounded-full px-4 py-2 text-sm focus:outline-none focus:border-pink-300"
            />
            <button type="submit" className="bg-pink-500 text-white p-2 rounded-full hover:bg-pink-600 transition-colors flex-shrink-0">
              <Send size={16} />
            </button>
          </form>
        </div>

        {/* Floating Button */}
        <button 
          onClick={() => setIsChatOpen(!isChatOpen)}
          className="bg-emerald-400 hover:bg-emerald-500 text-white p-4 rounded-full shadow-xl shadow-emerald-400/40 transition-all hover:scale-110 active:scale-95 flex items-center justify-center"
        >
          {isChatOpen ? <X size={28} /> : <MessageCircle size={28} />}
        </button>
      </div>

      {/* --- TOAST NOTIFICATION --- */}
      <div 
        className={`fixed bottom-10 left-1/2 transform -translate-x-1/2 z-50 flex items-center gap-3 bg-gray-900/90 backdrop-blur-md text-white px-6 py-4 rounded-full shadow-2xl transition-all duration-500 ease-out pointer-events-none ${
          toast.show ? 'translate-y-0 opacity-100' : 'translate-y-12 opacity-0'
        }`}
      >
        <CheckCircle2 className="text-emerald-400" size={24} />
        <span className="font-bold text-sm">{toast.message}</span>
      </div>

    </div>
  );
}
